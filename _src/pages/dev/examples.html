Plugin Guide & Examples

From SPI lifecycle to an end-to-end bridge mod: the code here comes from the official docs and the cfx-compat-epicfight example.

Lifecycle Callbacks

onInstallCompatConfigs

Called before onControlFlexReady. Use ICompatAssetInstaller.install() to install compat JSON into config/controlflex/compat/mods/ (higher priority than default/, lower than user/; same mod_id overrides wholesale). 

@Override
public void onInstallCompatConfigs(ICompatAssetInstaller installer) {
    installer.install(
        "/{{ROOT}}assets/mymod/compat/targetmod_keys.json",
        "targetmod_keys.json"
    );
}

onInstallGuideAssets

Same idea, but installs guide JSON into config/controlflex/guides/mods/. 

@Override
public void onInstallGuideAssets(ICompatAssetInstaller installer) {
    installer.install(
        "/{{ROOT}}assets/mymod/guides/targetmod_guid.json",
        "targetmod_guid.json"
    );
}

onControlFlexReady

The one initialization callback. All API providers are ready at this point. 

@Override
public void onControlFlexReady() {
    // 1. version check
    if (!requireApiVersion("0.8.7")) return;

    // 2. availability check
    if (!ControlFlexApi.isAvailable()) return;

    // 3. register your event listeners
    MinecraftForge.EVENT_BUS.addListener(this::onTargetModEvent);

    // 4. push initial state
    ControlFlexApi.getPlayerStateRegistry()
        .setState("mymod:initialized", true);
}

onControllerConnectionChanged

@Override
public void onControllerConnectionChanged(boolean connected) {
    if (connected) {
        String name = ControlFlexApi.getInputProvider().getGamepadName();
        LOGGER.info("Controller connected: {}", name);
    } else {
        LOGGER.info("Controller disconnected");
        resetState();
    }
}

Thread Safety & Logging

  • Except for IPlayerStateRegistry.setState() and IInputInjector, every API call must happen on the client main thread.
  • IControllerState and getActiveGameActions() are live views — do not cache references across ticks.
  • Log through your own logger:
private static final Logger LOGGER = LogManager.getLogger("my-mod-id");

Full Bridge Mod Template

package com.example.mymod;

import com.ifels.controlflex.api.*;
import org.apache.logging.log4j.*;

public class MyPlugin implements IControlFlexPlugin {

    private static final Logger LOGGER = LogManager.getLogger("my-bridge-mod");

    @Override
    public String getModId() {
        return "my_bridge_mod";
    }

    @Override
    public void onInstallCompatConfigs(ICompatAssetInstaller installer) {
        installer.install(
            "/{{ROOT}}assets/my_bridge_mod/compat/targetmod_keys.json",
            "targetmod_keys.json");
    }

    @Override
    public void onInstallGuideAssets(ICompatAssetInstaller installer) {
        installer.install(
            "/{{ROOT}}assets/my_bridge_mod/guides/targetmod_guid.json",
            "targetmod_guid.json");
    }

    @Override
    public void onControlFlexReady() {
        if (!requireApiVersion("0.8.7")) return;
        if (!ControlFlexApi.isAvailable()) return;

        IPlayerStateRegistry states = ControlFlexApi.getPlayerStateRegistry();
        states.setState("targetmod:initialized", true);

        LOGGER.info("Bridge mod ready. API version: {}",
            ControlFlexApi.getApiVersion());
    }

    @Override
    public void onControllerConnectionChanged(boolean connected) {
        LOGGER.info("Controller connection: {}", connected);
    }
}

Example 1: Reading the Movement Vector

public class ControllerInputReader {

    private static final float DEADZONE = 0.15f;

    /** Returns {forward, strafe}, each in -1.0..1.0 */
    public static float[] getMovement() {
        if (!ControlFlexApi.isAvailable()) return new float[]{0, 0};

        IInputProvider input = ControlFlexApi.getInputProvider();
        if (input == null || !input.isConnected()) return new float[]{0, 0};

        IControllerState state = input.getControllerState();
        if (state == null) return new float[]{0, 0};

        float forward = applyDeadzone(-state.getLeftStickY(), DEADZONE);
        float strafe  = applyDeadzone(state.getLeftStickX(), DEADZONE);

        return new float[]{forward, strafe};
    }

    private static float applyDeadzone(float v, float dz) {
        if (Math.abs(v) < dz) return 0;
        return Math.signum(v) * (Math.abs(v) - dz) / (1 - dz);
    }
}

Example 2: Checking Action States

public class ActionReader {

    public static boolean isAttacking() {
        IActionStateProvider actions = ControlFlexApi.getActionStateProvider();
        return actions != null && actions.isGameActionActive("attack");
    }

    public static boolean isModActionActive(String modId, String keyName) {
        IActionStateProvider actions = ControlFlexApi.getActionStateProvider();
        if (actions == null) return false;
        return actions.isGameActionActive(modId + ":" + keyName);
    }

    // Usage: ActionReader.isModActionActive("epicfight", "key.epicfight.dodge")
}

Example 3: Pushing Mod States (with compat JSON)

public class StatePusher {

    public static void onCombatModeChanged(boolean inCombat) {
        IPlayerStateRegistry registry = ControlFlexApi.getPlayerStateRegistry();
        if (registry != null) {
            registry.setState("mymod:combat_mode", inCombat);
        }
    }
}

Example 5: Controller-Specific Button Glyphs

public class GlyphHelper {

    public static String getAttackGlyph() {
        IInputProvider input = ControlFlexApi.getInputProvider();
        if (input == null) return "[Attack]";

        IControllerCapabilities caps = input.getCapabilities();
        if (caps == null) return "[A]";

        return switch (caps.getControllerType()) {
            case PLAYSTATION -> "[✕]";
            case XBOX         -> "[A]";
            case NINTENDO_SWITCH -> "[B]";
            default           -> "[A]";
        };
    }
}

Example 6: Declaring an Interactive Overlay (0.8.7)

Call these when your overlay opens or closes; Control Flex resolves the class name's stick behavior from compat configs. 

public class OverlayNotifier {

    public static void onOverlayOpened(Object overlay) {
        IInteractiveContextRegistrar registrar =
            ControlFlexApi.getInteractiveContextRegistrar();
        if (registrar != null) {
            registrar.notifyOverlayForeground(overlay.getClass().getName());
        }
    }

    public static void onOverlayClosed(Object overlay) {
        IInteractiveContextRegistrar registrar =
            ControlFlexApi.getInteractiveContextRegistrar();
        if (registrar != null) {
            registrar.notifyOverlayBackground(overlay.getClass().getName());
        }
    }
}

These were renamed from notifyForeground / notifyBackground in 0.8.7; mods compiled against 0.8.6 must be recompiled.
